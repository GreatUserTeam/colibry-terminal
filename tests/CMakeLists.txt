function(gus_terminal_test name)
    add_executable(${name} ${name}.cpp)
    target_link_libraries(${name} PRIVATE gus_terminal_core Qt6::Test)
    gus_terminal_warnings(${name})
    add_test(NAME ${name} COMMAND ${name})
    # Tests run without a display by default; export QT_QPA_PLATFORM=xcb (e.g. under xvfb-run)
    # at configure time to run them on a real platform plugin, including the OpenGL renderer.
    if(DEFINED ENV{QT_QPA_PLATFORM})
        set_tests_properties(${name} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=$ENV{QT_QPA_PLATFORM}" TIMEOUT 60)
    else()
        set_tests_properties(${name} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 60)
    endif()
endfunction()
